Question 069

Program to Iterate Over Characters in String

Solution

package strings;
public class Q069 {
    public static void main(String[] args) {
        String s = "Iterate";
        System.out.println("Characters:");
        for(int i=0;i<s.length();i++)
            System.out.println(s.charAt(i));
    }
}
