Question 018

Program to Find Largest Number in an array

Solution

package singlearrays;
public class Q018 {
    public static void main(String[] args) {
        int a[] = {34,91,17,68,85,52};
        int big = a[0];
        for(int i=0;i<a.length;i++){
            if(a[i]>big) big = a[i];
        }
        System.out.println("Largest = "+big);
    }
}
