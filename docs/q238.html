Question 238

Write a Java program to accept a float value of number and return a rounded float value

Solution

package basics;
import java.util.Scanner;
public class Q238 {
    static float roundKar(float x){
        return (float)Math.round(x);
    }
    public static void main(String[] args) {
        Scanner kb = new Scanner(System.in);
        System.out.print("Float value: ");
        float f = kb.nextFloat();
        System.out.println("Rounded = "+roundKar(f));
    }
}
