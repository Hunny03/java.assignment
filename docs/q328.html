Question 328

How to convert String to double

Solution

package conversion;
public class Q328 {
    public static void main(String[] args) {
        String s = "3.141592653589793";
        double d = Double.parseDouble(s);
        System.out.println("double = "+d);
        System.out.println("square = "+d*d);
    }
}
