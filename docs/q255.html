Question 255

Create a Java program to create a package with 4 public classes, create the object in another file and call the function created in those classes.

Solution

// ---- Q255.java ----
package packages;
import packages.q255.Addition;
import packages.q255.Subtraction;
import packages.q255.Multiplication;
import packages.q255.Division;
public class Q255 {
    public static void main(String[] args) {
        // objects dusre package ki classes ke, alag file me bana rahe hai
        Addition a = new Addition();
        Subtraction s = new Subtraction();
        Multiplication m = new Multiplication();
        Division d = new Division();
        System.out.println("Addition = "+a.kar(48,12));
        System.out.println("Subtraction = "+s.kar(48,12));
        System.out.println("Multiplication = "+m.kar(48,12));
        System.out.println("Division = "+d.kar(48,12));
    }
}

// ---- Addition.java ----
package packages.q255;
public class Addition {
    public int kar(int a, int b){ return a+b; }
}

// ---- Division.java ----
package packages.q255;
public class Division {
    public double kar(int a, int b){ return (double)a/b; }
}

// ---- Multiplication.java ----
package packages.q255;
public class Multiplication {
    public int kar(int a, int b){ return a*b; }
}

// ---- Subtraction.java ----
package packages.q255;
public class Subtraction {
    public int kar(int a, int b){ return a-b; }
}
