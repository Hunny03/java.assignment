Question 257

Create a Java program to create a package with public class and protected members to be accessed in another class.

Solution

// ---- Q257.java ----
package packages;
import packages.q257.Locker;
public class Q257 {
    // protected cheeze dusre package me bhi child class se mil jati hai
    static class Guard extends Locker {
        void dekho(){
            System.out.println("Guard checked the code: "+code);
            darwaza();
        }
    }
    public static void main(String[] args) {
        Guard g = new Guard();
        g.dekho();
    }
}

// ---- Locker.java ----
package packages.q257;
public class Locker {
    protected String code = "7391";
    protected void darwaza(){
        System.out.println("Locker opened with code "+code);
    }
}
