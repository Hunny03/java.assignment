Question 258

Create a Java program to create a nested package i.e. c4.sample with public class and protected members to be accessed in another class.

Solution

// ---- Q258.java ----
package packages;
import packages.c4.sample.Cabinet;      // package ke andar package: packages -> c4 -> sample
public class Q258 {
    static class Furniture extends Cabinet {
        void dikha(){
            drawers = drawers + 2;
            System.out.println(label());
        }
    }
    public static void main(String[] args) {
        Furniture f = new Furniture();
        f.dikha();
    }
}

// ---- Cabinet.java ----
package packages.c4.sample;
public class Cabinet {
    protected int drawers = 6;
    protected String label(){
        return "Cabinet jisme "+drawers+" drawers hai";
    }
}
