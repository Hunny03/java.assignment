Question 254

Create a Java program to create a package with multiple default classes.

Solution

// ---- Brake.java ----
package packages.q254;
class Brake {
    void dabao(){ System.out.println("The brakes were applied and the vehicle slowed down"); }
}

// ---- Engine.java ----
package packages.q254;
// public nahi likha isliye ye default class hai, sirf isi package me dikhegi
class Engine {
    void start(){ System.out.println("The engine started"); }
}

// ---- Q254.java ----
package packages.q254;
public class Q254 {
    public static void main(String[] args) {
        // teeno same package me hai isliye ek dusre ko dekh sakte hai
        Engine e = new Engine();
        Wheel w = new Wheel();
        Brake b = new Brake();
        e.start();
        w.ghoomo();
        b.dabao();
    }
}

// ---- Wheel.java ----
package packages.q254;
class Wheel {
    void ghoomo(){ System.out.println("The wheel is rotating"); }
}
