Question 259

Create a Java program to create a package with public class and public members to be accessed in another class.

Solution

// ---- Q259.java ----
package packages;
import packages.q259.Greeter;
public class Q259 {
    public static void main(String[] args) {
        Greeter g = new Greeter();
        g.name = "dusra package";      // public hai isliye kahin se bhi badal sakte hai
        g.sayHello();
    }
}

// ---- Greeter.java ----
package packages.q259;
public class Greeter {
    public String name = "Package World";
    public void sayHello(){
        System.out.println("Hello from "+name);
    }
}
