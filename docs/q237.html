Question 237

Write a Java program to convert a float value to absolute value

Solution

package basics;
public class Q237 {
    public static void main(String[] args) {
        float a = -48.65f;
        float b = 12.5f;
        System.out.println("abs("+a+") = "+Math.abs(a));
        System.out.println("abs("+b+") = "+Math.abs(b));
    }
}
