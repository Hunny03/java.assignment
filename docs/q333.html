Question 333

How to convert char to String

Solution

package conversion;
public class Q333 {
    public static void main(String[] args) {
        char c = 'A';
        String a = String.valueOf(c);
        String b = Character.toString(c);
        String d = c+"";
        System.out.println(a+" "+b+" "+d);
    }
}
