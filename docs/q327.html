Question 327

How to convert float to String

Solution

package conversion;
public class Q327 {
    public static void main(String[] args) {
        float f = 36.6f;
        String a = Float.toString(f);
        String b = String.valueOf(f);
        System.out.println(a);
        System.out.println(b);
    }
}
