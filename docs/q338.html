Question 338

How to convert int to double

Solution

package conversion;
public class Q338 {
    public static void main(String[] args) {
        int i = 47;
        double d = i;
        System.out.println("int = "+i);
        System.out.println("double = "+d);
        System.out.println("47/2 as an integer = "+(i/2));
        System.out.println("47/2 as a double = "+(d/2));
    }
}
