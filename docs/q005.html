Question 005

Program to print the elements of an array

Solution

package singlearrays;
public class Q005 {
    public static void main(String[] args) {
        int a[] = {1,2,5,10,20,50};
        System.out.println("Array elements:");
        for(int i=0;i<a.length;i++)
            System.out.print(a[i]+" ");
        System.out.println();
    }
}
