Question 266

Write a Java program to create a class known as Person with methods called getFirstName() and getLastName(). Create a subclass called Employee that adds a new method named getEmployeeId() and overrides the getLastName() method to include the employee's job title

Solution

package inheritance;
public class Q266 {
    static class Person {
        String firstName, lastName;
        Person(String f, String l){ firstName=f; lastName=l; }
        String getFirstName(){ return firstName; }
        String getLastName(){ return lastName; }
    }
    static class Employee extends Person {
        int id;
        String job;
        Employee(String f, String l, int id, String job){
            super(f,l);
            this.id = id;
            this.job = job;
        }
        int getId(){ return id; }
        String getLastName(){ return super.getLastName()+" ("+job+")"; }
    }
    public static void main(String[] args) {
        Employee e = new Employee("Sunil","Verma",4021,"Network Engineer");
        System.out.println("First name = "+e.getFirstName());
        System.out.println("Last name = "+e.getLastName());
        System.out.println("Id = "+e.getId());
    }
}
