Question 092

Program to display the lower triangular matrix

Solution

package multiarrays;
public class Q092 {
    public static void main(String[] args) {
        int a[][] = {{1,2,3},{4,5,6},{7,8,9}};
        System.out.println("Lower triangular matrix:");
        for(int i=0;i<3;i++){
            for(int j=0;j<3;j++){
                if(j<=i)
                    System.out.print(a[i][j]+"  ");
                else
                    System.out.print("0  ");
            }
            System.out.println();
        }
    }
}
