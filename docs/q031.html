Question 031

Write a Java method to display the current date and time.

Solution

package functions;
import java.time.LocalDateTime;
public class Q031 {
    static void abhi(){
        LocalDateTime t = LocalDateTime.now();
        System.out.println("Date and time: "+t);
    }
    public static void main(String[] args) {
        abhi();
    }
}
