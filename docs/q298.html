Question 298

Write a Java program to create a method that reads a file and throws an exception if the file is not found

Solution

package exceptions;
import java.io.*;
public class Q298 {
    static void padho(String name) throws FileNotFoundException {
        File f = new File(name);
        if(!f.exists())
            throw new FileNotFoundException("File nahi mili: "+name);
        System.out.println("File found");
    }
    public static void main(String[] args) {
        try{
            padho("koi_file_nahi.txt");
        }
        catch(FileNotFoundException e){
            System.out.println("Exception: "+e.getMessage());
        }
    }
}
