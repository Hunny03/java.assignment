Question 336

How to convert int to long

Solution

package conversion;
public class Q336 {
    public static void main(String[] args) {
        int i = 2000000000;
        long l = i;           // automatic hota hai
        System.out.println("int = "+i);
        System.out.println("long = "+l);
        System.out.println("long * 3 = "+l*3);
    }
}
